--!strict
--[[
    脚本名称: NoFog_Locker
    功能: 每帧持续锁定环境除雾 (清除传统雾效与大气层雾效)
    环境: Roblox Client (LocalScript)
    语言: Luau (开启严格类型检查)
]]

local Lighting: Lighting = game:GetService("Lighting")
local RunService: RunService = game:GetService("RunService")

-- 定义常量：设置一个超出游戏最大视距的极大值来消除 FogEnd 限制
local MAX_FOG_DISTANCE: number = 100000 

-- 缓存 Atmosphere 对象，避免每帧都去遍历查找，提升性能
local atmosphere: Atmosphere? = Lighting:FindFirstChildOfClass("Atmosphere")

-- 监听 Lighting 子对象变化，以防游戏动态生成 Atmosphere
Lighting.ChildAdded:Connect(function(child: Instance)
    if child:IsA("Atmosphere") then
        atmosphere = child
    end
end)

--[[
    核心除雾函数
]]
local function lockNoFog(): ()
    -- 1. 锁定传统 Lighting 雾效属性
    if Lighting.FogStart ~= 0 then
        Lighting.FogStart = 0
    end
    if Lighting.FogEnd ~= MAX_FOG_DISTANCE then
        Lighting.FogEnd = MAX_FOG_DISTANCE
    end
    
    -- 2. 锁定 Atmosphere (大气层) 的密度
    if atmosphere and atmosphere.Density ~= 0 then
        atmosphere.Density = 0
    end
end

-- 初始执行一次，立即生效
lockNoFog()

--[[
    每帧持续锁定
    使用 RenderStepped 确保在客户端每一帧渲染前强制覆盖雾效属性
]]
RunService.RenderStepped:Connect(lockNoFog)

print("[NoFog_Locker] 环境除雾已激活并锁定。")

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

-- 配置高亮属性
local HIGHLIGHT_COLOR = Color3.fromRGB(255, 0, 0) -- 纯红色
local FILL_TRANSPARENCY = 0.5 -- 填充透明度 (0为不透明，1为全透明)
local OUTLINE_TRANSPARENCY = 0 -- 轮廓透明度
local ENABLE_WALLHACK = true -- 是否开启穿墙透视 (true: 隔着墙也能看到, false: 被遮挡则不显示)

-- 核心函数：为指定角色添加高亮
local function applyHighlight(character)
	if not character then return end
	
	-- 防止重复添加（清理可能残留的旧高亮）
	local oldHighlight = character:FindFirstChild("CustomRedHighlight")
	if oldHighlight then
		oldHighlight:Destroy()
	end

	-- 创建新的高亮实例
	local highlight = Instance.new("Highlight")
	highlight.Name = "CustomRedHighlight"
	highlight.FillColor = HIGHLIGHT_COLOR
	highlight.OutlineColor = HIGHLIGHT_COLOR
	highlight.FillTransparency = FILL_TRANSPARENCY
	highlight.OutlineTransparency = OUTLINE_TRANSPARENCY
	
	-- 设置深度模式（是否穿墙可见）
	if ENABLE_WALLHACK then
		highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	else
		highlight.DepthMode = Enum.HighlightDepthMode.Occluded
	end
	
	-- 将高亮绑定到角色上
	highlight.Adornee = character
	highlight.Parent = character
end

-- 核心函数：处理单个玩家的监听逻辑
local function setupPlayer(player)
	-- 跳过本地玩家自己
	if player == LocalPlayer then return end

	-- 1. 如果玩家当前已经有角色，直接添加高亮
	if player.Character then
		applyHighlight(player.Character)
	end

	-- 2. 监听玩家重生 (CharacterAdded)
	player.CharacterAdded:Connect(function(character)
		-- 等待角色的核心部件加载完成，防止模型未加载完导致高亮失效
		character:WaitForChild("HumanoidRootPart", 5)
		-- 稍微延迟一帧，确保模型完全渲染
		task.wait(0.1) 
		applyHighlight(character)
	end)
end

-- ================= 运行逻辑 =================

-- 遍历当前已经在游戏里的玩家（防止脚本运行时漏掉已有的玩家）
for _, player in ipairs(Players:GetPlayers()) do
	setupPlayer(player)
end

-- 持续检测新玩家加入 (PlayerAdded)
Players.PlayerAdded:Connect(function(player)
	setupPlayer(player)
end)

	-- 这里不需要额外操作，因为 Player 和 Character 实例销毁时，其子对象 Highlight 也会自动被垃圾回收
end)

local RunService = game:GetService("RunService")

	return
end

	RunService.RenderStepped:Connect(function()

		end
	end)
else
	-- 如果不需要每帧刷新，仅设置一次
	humanoid.WalkSpeed = TARGET_SPEED
end
